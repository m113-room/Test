const SHEET_NAME_TASKS = "Tasks"; const SHEET_NAME_EVENTS = "Events"; const SHEET_NAME_OPTIONS = "Options"; const SHEET_NAME_USERS = "Users"; const SHEET_NAME_LOGS = "Logs"; function doGet(e) { try { var action = e && e.parameter ? e.parameter.action : ''; var ss = SpreadsheetApp.getActiveSpreadsheet(); if (action === 'checkLogin') { var username = e.parameter.username; var sheet = ss.getSheetByName(SHEET_NAME_USERS); var data = sheet.getDataRange().getValues(); var foundUser = null; for (var i = 1; i < data.length; i++) { if (String(data[i][0]).trim() === String(username).trim()) { foundUser = { username: data[i][0], role: data[i][1], name: data[i][2] }; break; } } return ContentService.createTextOutput(JSON.stringify({ success: foundUser !== null, user: foundUser })).setMimeType(ContentService.MimeType.JSON); } if (action === 'getEvents') { var sheet = ss.getSheetByName(SHEET_NAME_EVENTS); var data = sheet.getDataRange().getValues(); var events = {}; for (var i = 1; i < data.length; i++) { if (data[i][0]) { var dateKey = Utilities.formatDate(new Date(data[i][0]), Session.getScriptTimeZone(), "yyyy-MM-dd"); if (!events[dateKey]) events[dateKey] = []; events[dateKey].push(data[i][1]); } } return ContentService.createTextOutput(JSON.stringify(events)).setMimeType(ContentService.MimeType.JSON); } if (action === 'getOptions') { var sheet = ss.getSheetByName(SHEET_NAME_OPTIONS); var data = sheet.getDataRange().getValues(); var codes = []; var subjects = []; for (var i = 1; i < data.length; i++) { if (data[i][0]) codes.push(String(data[i][0])); if (data[i][1]) subjects.push(String(data[i][1])); } return ContentService.createTextOutput(JSON.stringify({ codes: codes, subjects: subjects })).setMimeType(ContentService.MimeType.JSON); } if (action === 'getLogs') { var all = e.parameter.all === 'true'; var sheet = ss.getSheetByName(SHEET_NAME_LOGS); var data = sheet.getDataRange().getValues(); var logs = []; for (var i = 1; i < data.length; i++) { logs.push({ username: data[i][0], actionText: data[i][1], details: data[i][2], timestamp: data[i][3] }); } if (!all && logs.length > 20) { logs = logs.slice(logs.length - 20); } return ContentService.createTextOutput(JSON.stringify(logs)).setMimeType(ContentService.MimeType.JSON); } var sheet = ss.getSheetByName(SHEET_NAME_TASKS); var data = sheet.getDataRange().getValues(); var tasks = []; for (var i = 1; i < data.length; i++) { if (data[i][0]) { tasks.push({ id: String(data[i][0]), code: data[i][1], subject: data[i][2], detail: data[i][3], assignDate: data[i][4] ? Utilities.formatDate(new Date(data[i][4]), Session.getScriptTimeZone(), "yyyy-MM-dd") : '', dueDate: data[i][5] ? Utilities.formatDate(new Date(data[i][5]), Session.getScriptTimeZone(), "yyyy-MM-dd") : '', isPinned: data[i][6] === true || data[i][6] === "TRUE", submitted: data[i][7] ? String(data[i][7]).split(",") : [], extraUnsubmitted: data[i][8] ? String(data[i][8]).split(",") : [] }); } } return ContentService.createTextOutput(JSON.stringify(tasks)).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ error: error.message })).setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { try { var data = JSON.parse(e.postData.contents); var action = data.action; var ss = SpreadsheetApp.getActiveSpreadsheet(); if (action === 'saveTasks') { var sheet = ss.getSheetByName(SHEET_NAME_TASKS); sheet.clearContents(); sheet.appendRow(["ID", "Code", "Subject", "Detail", "AssignDate", "DueDate", "IsPinned", "Submitted", "ExtraUnsubmitted"]); var tasks = data.tasks || []; tasks.forEach(function(t) { sheet.appendRow([ t.id, t.code, t.subject, t.detail, t.assignDate, t.dueDate, t.isPinned, (t.submitted || []).join(","), (t.extraUnsubmitted || []).join(",") ]); }); } else if (action === 'saveEvents') { var sheet = ss.getSheetByName(SHEET_NAME_EVENTS); sheet.clearContents(); sheet.appendRow(["Date", "Event"]); var events = data.events || {}; Object.keys(events).forEach(function(date) { (events[date] || []).forEach(function(ev) { sheet.appendRow([date, ev]); }); }); } else if (action === 'saveOption') { var sheet = ss.getSheetByName(SHEET_NAME_OPTIONS); if (data.code || data.subject) { sheet.appendRow([data.code || "", data.subject || ""]); } } else if (action === 'saveAllOptions') { var sheet = ss.getSheetByName(SHEET_NAME_OPTIONS); sheet.clearContents(); // ล้างข้อมูลเก่าทั้งหมดในชีต Options ทิ้งก่อน sheet.appendRow(["Code", "Subject"]); // ใส่หัวตารางใหม่ var codes = data.codes || []; var subjects = data.subjects || []; var maxLen = Math.max(codes.length, subjects.length); for (var i = 0; i < maxLen; i++) { sheet.appendRow([ codes[i] || "", subjects[i] || "" ]); } } else if (action === 'saveLog') { var sheet = ss.getSheetByName(SHEET_NAME_LOGS); sheet.appendRow([ data.username || "Unknown", data.actionText || "", data.details || "", new Date() ]); } else if (action === 'clearLogs') { var sheet = ss.getSheetByName(SHEET_NAME_LOGS); sheet.clearContents(); sheet.appendRow(["Username", "Action", "Details", "Timestamp"]); } return ContentService.createTextOutput(JSON.stringify({ success: true })).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ success: false, error: error.message })).setMimeType(ContentService.MimeType.JSON); } }
